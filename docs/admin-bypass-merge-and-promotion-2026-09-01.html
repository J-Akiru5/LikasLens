export async function bulkTicketStatus(ids: string[], status: string) {
  const failed: string[] = [];
  for (const id of ids) {
    try {
      await routeTicketStatus(id, status);

  return {
    success: failed.length === 0,

Why this side is correct: Development's version attempted the AI proxy per-ticket, but on rejection fell back to a raw, unauthorized db().from("tickets").update(...) — no transition validation, no timeline entry. This is the same class of bypass the entire fix exists to close. The fix branch version routes every ticket through the shared routeTicketStatus() helper, which uses the authorized proxy first and falls back with a best-effort timeline entry only when the AI service is unreachable.
